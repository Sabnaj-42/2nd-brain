    ```

## DB2 HADR configuration commands
```bash
# 1. Display all database configuration parameters for the database named 'abc'
db2 get db cfg for abc  # Shows DB configuration, including HADR, logging, memory, and isolation levels

# 2. Set the isolation level on the standby database to Uncommitted Read (UR)
export DB2_STANDBY_ISO=UR  # Allows read-only queries on the standby (for ROS). Must be set before activating standby.

# 3. Enable Read on Standby (ROS) functionality on the standby
export DB2_HADR_ROS=ON  # Enables the standby database to accept queries while HADR replication is active

# 4. Control whether index build operations are logged (applies to PRIMARY database)
db2 update db cfg for abc using LOGINDEXBUILD ON  # ON: logs index builds (safer); OFF: skips logging (faster). 
```
### Configuration on Primary

```bash
db_name=abc
db2 create database $db_name
db2 connect to db_name
#hostname -f // running this command in primary pod gives the hostname of primary pod
hostL=db2-0.db2-service.db2-system.svc.cluster.local
#hostname -f // running this command in standby pod gives the hostname of standby pod
hostR=db2-1.db2-service.db2-system.svc.cluster.local
# ================================
# 1. Configure all HADR parameters
# ================================

# Set log archiving method (required for HADR)
db2 UPDATE DB CFG FOR $db_name USING LOGARCHMETH1 "DISK:/database/backup" #DISK:/database/backup directory must exist or created. (pvc mounted path can be used)

# Log index build operations for safe replication
db2 UPDATE DB CFG FOR $db_name USING LOGINDEXBUILD ON

# Recover indexes automatically after failover/restart
db2 UPDATE DB CFG FOR $db_name USING INDEXREC RESTART

# Primary database HADR local host (this pod's hostname)
db2 UPDATE DB CFG FOR $db_name USING HADR_LOCAL_HOST $hostL  #hostname is generated by "<pod-name>.<service-name>.<namespace>.svc.cluster.local"


# Primary HADR listener port
db2 UPDATE DB CFG FOR $db_name USING HADR_LOCAL_SVC 55000

# Standby database hostname (remote host)
db2 UPDATE DB CFG FOR $db_name USING HADR_REMOTE_HOST $hostR

# Standby HADR port (must match standby's HADR_LOCAL_SVC)
db2 UPDATE DB CFG FOR $db_name USING HADR_REMOTE_SVC 55000

# Remote DB2 instance name on standby
db2 UPDATE DB CFG FOR $db_name USING HADR_REMOTE_INST db2inst1

# Synchronization mode (NEARSYNC = balanced perf + safety)
db2 UPDATE DB CFG FOR $db_name USING HADR_SYNCMODE NEARSYNC

# HADR timeout in seconds for network failures
db2 UPDATE DB CFG FOR $db_name USING HADR_TIMEOUT 120


# ================================
# 2. Restart the database
# ================================

# Deactivate database to apply config
db2 deactivate db $db_name


# ================================
# 3. Take a full backup for standby
# ================================

# Backup database to a shared directory accessible by the standby
db2 BACKUP DATABASE $db_name TO "/database/backup" COMPRESS

db2 terminate

# Activate database again
db2 activate db $db_name


# ================================
# 4. Start HADR on primary
# ================================

# Start HADR (only after standby is fully configured and in PEER_SYNC mode)
db2 START HADR ON DB $db_name AS PRIMARY

```

### Configuration on Standby
```bash
db_name=abc
#hostname -f // running this command in standby pod gives the hostname of standby pod
hostL=db2-1.db2-service.db2-system.svc.cluster.local
#hostname -f // running this command in primary pod gives the hostname of primary pod
hostR=db2-0.db2-service.db2-system.svc.cluster.local

# ===========================================
# STANDBY POD HADR CONFIGURATION SCRIPT
# ===========================================
# EXECUTE AFTER PRIMARY BACKUP IS COMPLETE AND ACCESSIBLE

# ================================
# 0. Set DB2 Registry Variables
# ================================
# Enable read-only access on standby (Hot Standby)
db2set DB2_HADR_ROS=ON

# Set default isolation level to Uncommitted Read for standby connections
db2set DB2_STANDBY_ISO=UR

# Apply registry changes (requires instance restart if not already set)
# db2stop force
# db2start

# ================================
# 1. Verify and Prepare Environment
# ================================
# Check for existing databases (abc should NOT exist before restore)
db2 LIST DATABASE DIRECTORY

# IMPORTANT: DO NOT run "db2 CREATE DATABASE" on standby.
# The standby MUST be created by restoring the primary's backup.

# Ensure primary's backup is accessible at the shared location
# Backup from primary should be at: /database/backup/

# ================================
# 2. Restore Database from Primary Backup
# ================================
# Restore database from the shared backup location
# Replace WITH <timestamp> if multiple backups exist
db2 RESTORE DATABASE $db_name FROM "/database/backup" 

# ================================
# 3. Configure HADR Parameters on Standby
# ================================
# Log index build operations (must match primary)
db2 UPDATE DB CFG FOR $db_name USING LOGINDEXBUILD ON

# Automatic index recovery (must match primary)
db2 UPDATE DB CFG FOR $db_name USING INDEXREC RESTART

# Standby HADR local host (this pod's hostname)
db2 UPDATE DB CFG FOR $db_name USING HADR_LOCAL_HOST $hostL

# Standby HADR listener port (must match primary's HADR_REMOTE_SVC)
db2 UPDATE DB CFG FOR $db_name USING HADR_LOCAL_SVC 55000

# Primary database hostname (remote host)
db2 UPDATE DB CFG FOR $db_name USING HADR_REMOTE_HOST $hostR

# Primary HADR port (must match primary's HADR_LOCAL_SVC) 
db2 UPDATE DB CFG FOR $db_name USING HADR_REMOTE_SVC 55000

# Remote instance name on primary
db2 UPDATE DB CFG FOR $db_name USING HADR_REMOTE_INST db2inst1

# Synchronization mode (must match primary)
db2 UPDATE DB CFG FOR $db_name USING HADR_SYNCMODE NEARSYNC

# Replay delay (0 = no delay, recommended for immediate failover)
db2 UPDATE DB CFG FOR $db_name USING HADR_REPLAY_DELAY 0

# HADR timeout (should match primary)
db2 UPDATE DB CFG FOR $db_name USING HADR_TIMEOUT 120

# Log archiving method (must match primary)
#db2 UPDATE DB CFG FOR $db_name USING LOGARCHMETH1 "DISK:/database/backup" #optional for standby database

# ================================
# 4. Start HADR on Standby
# ================================
# Start HADR in standby role (ALWAYS start standby before primary)
db2 START HADR ON DATABASE $db_name AS STANDBY

# Verify standby status
db2pd -db $db_name -hadr

# ================================
# 5. Startup Sequence Reminder
# ================================
# AFTER standby shows PEER or REMOTE_CATCHUP state,
# return to PRIMARY pod and execute:
# db2 START HADR ON DB HADRDB AS PRIMARY
#
# Monitor both servers:

## Command for taking backup using pipeline and cat it in another terminal

1. Create pipe and Start restore from it (will block) -- in standby pod
```bash
  mkfifo /tmp/db2backup.pipe 
  db2 restore database abc from /tmp/db2backup.pipe
```
2. Create pipe and start taking backup into this pipe -- in Primary pod
```bash
mkfifo /tmp/db2backup.pipe
db2 backup database abc to /tmp/db2backup.pipe
```
3. Stream data from primary pod pipe and write it into standby pod pipe -- in another terminal
```bash
kubectl exec my-db2-0 -n demo -- \
              su - db2inst1 -c "cat /tmp/db2backup.pipe" \
            | kubectl exec -i my-db2-1 -n demo -- \
              su - db2inst1 -c "cat > /tmp/db2backup.pipe"
```

## 3 Replica
### On Primary pod
```bash
db2set DB2_STANDBY_ISO=UR
db2set DB2_HADR_ROS=ON
db_name=abc
db2 LIST DATABASE DIRECTORY # list databases
db2 CREATE DATABASE $db_name # (IF NOT EXISTS ANY db)

host=my-db2-0.my-db2-pods.demo.svc.cluster.local

db2 UPDATE DB CFG FOR $db_name USING LOGINDEXBUILD     ON
db2 UPDATE DB CFG FOR $db_name USING INDEXREC          RESTART
db2 UPDATE DB CFG FOR $db_name USING HADR_LOCAL_HOST   $host
db2 UPDATE DB CFG FOR $db_name USING HADR_LOCAL_SVC    55005
db2 UPDATE DB CFG FOR $db_name USING HADR_REMOTE_HOST  my-db2-1.my-db2-pods.demo.svc.cluster.local
db2 UPDATE DB CFG FOR $db_name USING HADR_REMOTE_SVC   55005
db2 UPDATE DB CFG FOR $db_name USING HADR_REMOTE_INST  db2inst1
db2 UPDATE DB CFG FOR $db_name USING HADR_TARGET_LIST  "my-db2-1.my-db2-pods.demo.svc.cluster.local:55005|my-db2-2.my-db2-pods.demo.svc.cluster.local:55005"
db2 UPDATE DB CFG FOR $db_name USING HADR_SYNCMODE     NEARSYNC
db2 UPDATE DB CFG FOR $db_name USING HADR_REPLAY_DELAY 0
db2 UPDATE DB CFG FOR $db_name USING HADR_TIMEOUT      120
db2 UPDATE DB CFG FOR $db_name USING LOGARCHMETH1      "DISK:/shared"

db2 BACKUP DATABASE $db_name TO "/tmp" compress # I have use pipe to backup and restore database

db2 terminate

# Activate database again
db2 activate db $db_name


# ================================
# 4. Start HADR on primary
# ================================

# Start HADR (only after standby is fully configured and in PEER_SYNC mode)
# At first start HADR on all standby pod  and then start HADR on primary pod
db2 START HADR ON DB $db_name AS PRIMARY
```

### Backup resotre by copying in local machine
1. Backup database in primary pod
```bash
db2 backup db ABC to /database/backup compress
```
2. Copy primay pod to local machine , then local machine to standby pod
```bash
kubectl exec -n demo my-db2-0 -- \
        cat /database/backup/ABC.0.db2inst1.DBPART000.20260213082632.001 \
      | kubectl exec -i -n demo my-db2-2 -- \
        tee /database/backup/ABC.0.db2inst1.DBPART000.20260213082632.001 > /dev/null
```
3. Restore database in standby pod
```bash
db2 restore database ABC from /database/backup taken at 20260213082632
```
