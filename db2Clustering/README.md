## DB2 HADR configuration commands
```bash
# 1. Display all database configuration parameters for the database named 'abc'
db2 get db cfg for abc  # Shows DB configuration, including HADR, logging, memory, and isolation levels

# 2. Set the isolation level on the standby database to Uncommitted Read (UR)
export DB2_STANDBY_ISO=UR  # Allows read-only queries on the standby (for ROS). Must be set before activating standby.

# 3. Enable Read on Standby (ROS) functionality on the standby
export DB2_HADR_ROS=ON  # Enables the standby database to accept queries while HADR replication is active

# 4. Control whether index build operations are logged (applies to PRIMARY database)
db2 update db cfg for abc using LOGINDEXBUILD ON  # ON: logs index builds (safer); OFF: skips logging (faster). 
```
### Configuration on Primary

```bash
db_name=abc

# ================================
# 1. Configure all HADR parameters
# ================================

# Set log archiving method (required for HADR)
db2 UPDATE DB CFG FOR $db_name USING LOGARCHMETH1 "DISK:/shared-logs/" #DISK:/shared-logs directory must exist or created. (pvc mounted path)

# Log index build operations for safe replication
db2 UPDATE DB CFG FOR $db_name USING LOGINDEXBUILD ON

# Recover indexes automatically after failover/restart
db2 UPDATE DB CFG FOR $db_name USING INDEXREC RESTART

# Primary database HADR local host (this pod's hostname)
db2 UPDATE DB CFG FOR $db_name USING HADR_LOCAL_HOST db2-5f989464fb-cnvct.db2-service.db2-system-test.svc.cluster.local  #hostname is generated by "<pod-name>.<service-name>.<namespace>.svc.cluster.local"


# Primary HADR listener port
db2 UPDATE DB CFG FOR $db_name USING HADR_LOCAL_SVC 55000

# Standby database hostname (remote host)
db2 UPDATE DB CFG FOR $db_name USING HADR_REMOTE_HOST db2-5f989464fb-5prh2.db2-service.db2-system-test.svc.cluster.local

# Standby HADR port (must match standby's HADR_LOCAL_SVC)
db2 UPDATE DB CFG FOR $db_name USING HADR_REMOTE_SVC 55000

# Remote DB2 instance name on standby
db2 UPDATE DB CFG FOR $db_name USING HADR_REMOTE_INST db2inst1

# Synchronization mode (NEARSYNC = balanced perf + safety)
db2 UPDATE DB CFG FOR $db_name USING HADR_SYNCMODE NEARSYNC

# HADR timeout in seconds for network failures
db2 UPDATE DB CFG FOR $db_name USING HADR_TIMEOUT 120


# ================================
# 2. Restart the database
# ================================

# Deactivate database to apply config
db2 deactivate db $db_name

# Activate database again
db2 activate db $db_name


# ================================
# 3. Take a full backup for standby
# ================================

# Backup database to a shared directory accessible by the standby
db2 BACKUP DATABASE $db_name TO "/shared-backups/" COMPRESS


# ================================
# 4. Start HADR on primary
# ================================

# Start HADR (only after standby is fully configured and in PEER_SYNC mode)
db2 START HADR ON DB $db_name AS PRIMARY

```
